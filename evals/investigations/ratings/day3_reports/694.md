```sql
WITH cohort_2025_accounts AS (
    SELECT
        am.account_id,
        MAX(CASE WHEN am.is_churn_month THEN 1 ELSE 0 END) AS ever_churned_flag
    FROM saasco_marts.mart_account_months am
    WHERE am.cohort_month >= DATE '2025-01-01'
      AND am.cohort_month < DATE '2026-01-01'
    GROUP BY am.account_id

account_churn_flag AS (
    SELECT
        ca.account_id,
        CASE WHEN ca.ever_churned_flag = 1 THEN 'true' ELSE 'false' END AS eventually_churned
    FROM cohort_2025_accounts ca
),
tenure_windows AS (
    SELECT
        am.account_id,
        am.month AS tenure_calendar_month
    FROM saasco_marts.mart_account_months am
    WHERE am.cohort_month >= DATE '2025-01-01'
      AND am.cohort_month < DATE '2026-01-01'
      AND am.tenure_month BETWEEN 2 AND 6
),
weekly_in_tenure AS (
    SELECT
        tw.account_id,
        u.week_start,
        u.key_actions
    FROM tenure_windows tw
    JOIN saasco_staging.stg_usage_weekly u
        ON u.account_id = tw.account_id

        AND u.week_start < (tw.tenure_calendar_month + INTERVAL '1 month')
),
per_account_avg AS (
    SELECT
        wit.account_id,
        AVG(wit.key_actions) AS avg_kactions
    FROM weekly_in_tenure wit
    GROUP BY wit.account_id
),
combined AS (
    SELECT
        acf.account_id AS combined_account_id,
        acf.eventually_churned AS combined_eventually_churned,

    LEFT JOIN per_account_avg paa
        ON paa.account_id = acf.account_id
)
SELECT
```

```sql
WITH cohort_base AS (
    SELECT
        CAST(YEAR(start_date) AS VARCHAR) AS cohort_year,
        plan_tier,
        billing_interval,
        account_id
    FROM saasco_marts.mart_subscription_lifecycle
    WHERE YEAR(start_date) IN (2024, 2025)
),
cohort_totals AS (
    SELECT
        cb.cohort_year,
        COUNT(DISTINCT cb.account_id) AS total_accounts
    FROM cohort_base cb
    GROUP BY cb.cohort_year
),
cohort_segments AS (
    SELECT
        cb.cohort_year,
        cb.plan_tier,
        cb.billing_interval,
        COUNT(DISTINCT cb.account_id) AS n_accounts
    FROM cohort_base cb
    GROUP BY cb.cohort_year, cb.plan_tier, cb.billing_interval
)
SELECT
    s.cohort_year,
    s.plan_tier,
    s.billing_interval,
    s.n_accounts,
    ROUND(s.n_accounts * 100.0 / t.total_accounts, 2) AS pct_of_cohort
FROM cohort_segments s
JOIN cohort_totals t ON s.cohort_year = t.cohort_year
ORDER BY s.cohort_year, s.plan_tier, s.billing_interval
```

```sql
SELECT

    COUNT(*) FILTER (WHERE is_churn_month = TRUE) * 1.0 / COUNT(*) AS churn_rate
FROM saasco_marts.mart_account_months
WHERE YEAR(cohort_month) IN (2024, 2025)
GROUP BY
    plan_tier,
    billing_interval,
    YEAR(cohort_month)
ORDER BY
```
