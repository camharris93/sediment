The core issue: payment infrastructure degradation in early 2026 is driving both revenue leakage (13.04% failure rate) and customer churn, creating a feedback loop that threatens retention and MRR.

```sql
WITH base AS (
    SELECT
        account_id,
        CASE
            WHEN month >= DATE '2026-01-01' AND month <= DATE '2026-06-01' THEN 'early_2026'
            WHEN month >= DATE '2025-07-01' AND month <= DATE '2025-12-01' THEN 'late_2025_baseline'
        END AS period,
        mrr,
        active_users_avg,
        failed_attempts_30d,
        is_churn_month
    FROM saasco_marts.mart_account_months
    WHERE
        (month >= DATE '2026-01-01' AND month <= DATE '2026-06-01')
        OR (month >= DATE '2025-07-01' AND month <= DATE '2025-12-01')
)
SELECT
    account_id,
    period,
    AVG(mrr)                                        AS avg_mrr,
    AVG(active_users_avg)                           AS avg_active_users,
    AVG(CAST(failed_attempts_30d AS DOUBLE))        AS avg_failed_attempts,
    CAST(MAX(CASE WHEN is_churn_month THEN 1 ELSE 0 END) AS BOOLEAN) AS churn_flag
FROM base
WHERE period IS NOT NULL
GROUP BY account_id, period
ORDER BY account_id, period
```

```sql
WITH invoice_monthly AS (
    SELECT
        DATE_TRUNC('month', invoice_date) AS inv_month,
        COUNT(*) AS total_invoices,
        COUNT(*) FILTER (WHERE status = 'failed') AS failed_invoices,
        ROUND(
            COUNT(*) FILTER (WHERE status = 'failed') * 1.0 / NULLIF(COUNT(*), 0),
            4
        ) AS payment_failure_rate
    FROM saasco_staging.stg_invoices
    WHERE invoice_date >= '2025-10-01'
      AND invoice_date < '2026-04-01'
    GROUP BY DATE_TRUNC('month', invoice_date)
),
ticket_monthly AS (
    SELECT
        DATE_TRUNC('month', created_at) AS tkt_month,
        COUNT(*) AS total_tickets,
        COUNT(*) FILTER (WHERE priority IN ('urgent', 'high')) AS urgent_high_tickets,
        ROUND(
            COUNT(*) FILTER (WHERE priority IN ('urgent', 'high')) * 1.0 / NULLIF(COUNT(*), 0),
            4
        ) AS urgent_high_share,
        AVG(csat) FILTER (WHERE csat IS NOT NULL) AS avg_csat
    FROM saasco_staging.stg_support_tickets
    WHERE created_at >= '2025-10-01'
      AND created_at < '2026-04-01'
    GROUP BY DATE_TRUNC('month', created_at)
),
combined AS (
    SELECT
        COALESCE(i.inv_month, t.tkt_month) AS combined_month,
        i.total_invoices,
        i.failed_invoices,
        i.payment_failure_rate,
        t.total_tickets,
        t.urgent_high_tickets,
        t.urgent_high_share,
        ROUND(t.avg_csat, 3) AS avg_csat
    FROM invoice_monthly i
    FULL OUTER JOIN ticket_monthly t ON i.inv_month = t.tkt_month
)
SELECT
    c.combined_month AS month,
    CASE
        WHEN c.combined_month >= '2026-01-01' THEN 'early_2026'
        ELSE 'late_2025_baseline'
    END AS comparison_period,
    c.total_invoices,
    c.failed_invoices,
    c.payment_failure_rate,
    c.total_tickets,
    c.urgent_high_tickets,
    c.urgent_high_share,
    c.avg_csat
FROM combined c
ORDER BY c.combined_month
```
