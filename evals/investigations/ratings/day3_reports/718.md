WITH latest_sub AS (
    SELECT
        account_id,
        plan_tier,

        a.account_id AS account_id,
        a.signup_date AS signup_date,

            CAST(QUARTER(a.signup_date) AS VARCHAR),
            ' ',
            CAST(YEAR(a.signup_date) AS VARCHAR)

    INNER JOIN latest_sub AS s
        ON a.account_id = s.account_id
        AND s.rn = 1
    WHERE a.signup_date >= DATE '2025-01-01'
      AND a.signup_date <= DATE '2026-06-30'
),
cell_counts AS (

        account_sub.signup_quarter_date AS signup_quarter_date,
        account_sub.plan_tier AS plan_tier,
        account_sub.billing_interval AS billing_interval,
        COUNT(DISTINCT account_sub.account_id) AS new_account_count
    FROM account_sub
    GROUP BY

        account_sub.plan_tier,

```sql
WITH churned_accounts AS (
    SELECT
        account_id,
        CASE
            WHEN month BETWEEN DATE '2026-04-01' AND DATE '2026-06-30' THEN 'Q2 2026'
            WHEN month BETWEEN DATE '2026-01-01' AND DATE '2026-03-31' THEN 'Q1 2026'
            WHEN month BETWEEN DATE '2025-04-01' AND DATE '2025-06-30' THEN 'Q2 2025'
        END AS churn_quarter
    FROM saasco_marts.mart_account_months
    WHERE is_churn_month = TRUE
      AND (
          month BETWEEN DATE '2026-04-01' AND DATE '2026-06-30'
          OR month BETWEEN DATE '2026-01-01' AND DATE '2026-03-31'
          OR month BETWEEN DATE '2025-04-01' AND DATE '2025-06-30'
      )
),
early_usage AS (
    SELECT
        am.account_id,
        am.active_users_avg
    FROM saasco_marts.mart_account_months am
    WHERE am.tenure_month IN (1, 2, 3)
)
SELECT
    ca.churn_quarter,
    AVG(eu.active_users_avg) AS avg_early_active_users,
    COUNT(DISTINCT ca.account_id) AS churned_account_count
FROM churned_accounts ca
JOIN early_usage eu ON ca.account_id = eu.account_id
GROUP BY ca.churn_quarter
ORDER BY
    CASE ca.churn_quarter
        WHEN 'Q2 2025' THEN 1
        WHEN 'Q1 2026' THEN 2
        WHEN 'Q2 2026' THEN 3
    END

WITH filtered AS (
    SELECT
        cancel_reason,
        end_date,
        CASE
            WHEN end_date BETWEEN DATE '2025-04-01' AND DATE '2025-06-30' THEN 'Q2 2025'
            WHEN end_date BETWEEN DATE '2026-01-01' AND DATE '2026-03-31' THEN 'Q1 2026'
            WHEN end_date BETWEEN DATE '2026-04-01' AND DATE '2026-06-30' THEN 'Q2 2026'
        END AS end_quarter
    FROM saasco_staging.stg_subscriptions
    WHERE status = 'canceled'
      AND end_date IS NOT NULL
      AND cancel_reason IS NOT NULL
      AND (
          end_date BETWEEN DATE '2025-04-01' AND DATE '2025-06-30'
          OR end_date BETWEEN DATE '2026-01-01' AND DATE '2026-03-31'
          OR end_date BETWEEN DATE '2026-04-01' AND DATE '2026-06-30'
      )
),
quarter_totals AS (
    SELECT
        f.end_quarter,
        COUNT(*) AS total_churned_that_quarter
    FROM filtered f
    GROUP BY f.end_quarter
),
reason_counts AS (
    SELECT
        f.end_quarter,
        f.cancel_reason,
        COUNT(*) AS churned_subscription_count
    FROM filtered f
    GROUP BY f.end_quarter, f.cancel_reason
)
SELECT
    rc.end_quarter,
    rc.cancel_reason,
    rc.churned_subscription_count,
    qt.total_churned_that_quarter
FROM reason_counts rc
JOIN quarter_totals qt ON rc.end_quarter = qt.end_quarter
ORDER BY
    CASE rc.end_quarter
        WHEN 'Q2 2025' THEN 1
        WHEN 'Q1 2026' THEN 2
        WHEN 'Q2 2026' THEN 3
    END,
    rc.churned_subscription_count DESC

WITH active_month_accounts AS (
    SELECT
        month,
        account_id
    FROM saasco_marts.mart_account_months
    WHERE month >= DATE '2025-10-01'
      AND month <= DATE '2026-06-01'
),

active_account_counts AS (
    SELECT
        month,
        COUNT(DISTINCT account_id) AS active_accounts
    FROM active_month_accounts
    GROUP BY month
),

ticket_months AS (
    SELECT
        t.account_id,
        DATE_TRUNC('month', t.created_at) AS ticket_month,
        t.csat
    FROM saasco_staging.stg_support_tickets AS t
    WHERE DATE_TRUNC('month', t.created_at) >= DATE '2025-10-01'
      AND DATE_TRUNC('month', t.created_at) <= DATE '2026-06-01'
),

ticket_aggs AS (
    SELECT
        tm.ticket_month,
        COUNT(*) AS total_tickets,
        AVG(tm.csat) AS avg_csat
    FROM ticket_months AS tm
    INNER JOIN active_month_accounts AS ama
        ON ama.account_id = tm.account_id
        AND ama.month = tm.ticket_month
    GROUP BY tm.ticket_month
)

SELECT
    aac.month,
    ROUND(ta.avg_csat, 4)                                      AS avg_csat,
    ROUND(CAST(ta.total_tickets AS DOUBLE) / aac.active_accounts, 4) AS tickets_per_active_account,
    COALESCE(ta.total_tickets, 0)                              AS total_tickets,
    aac.active_accounts
FROM active_account_counts AS aac
LEFT JOIN ticket_aggs AS ta
    ON ta.ticket_month = aac.month
ORDER BY aac.month
```

WITH churned_accounts AS (
    SELECT
        msl.account_id,
        MAX(msl.end_date) AS ref_end_date
    FROM saasco_marts.mart_subscription_lifecycle msl
    WHERE msl.ended_by = 'churn'
      AND msl.end_date BETWEEN DATE '2026-04-01' AND DATE '2026-06-30'
    GROUP BY msl.account_id
),
retained_accounts AS (
    SELECT DISTINCT msl.account_id
    FROM saasco_marts.mart_subscription_lifecycle msl
    WHERE (msl.end_date IS NULL OR msl.end_date > DATE '2026-06-30')
      AND msl.account_id NOT IN (SELECT ca.account_id FROM churned_accounts ca)
      AND msl.account_id IN (
          SELECT am.account_id
          FROM saasco_marts.mart_account_months am
          WHERE am.month BETWEEN DATE '2026-04-01' AND DATE '2026-06-01'
            AND am.is_churn_month = FALSE
      )
),
churned_ticket_counts AS (
    SELECT
        ca.account_id,
        COUNT(DISTINCT st.ticket_id) AS ticket_count
    FROM churned_accounts ca
    LEFT JOIN saasco_staging.stg_support_tickets st
        ON st.account_id = ca.account_id
        AND st.created_at >= CAST(ca.ref_end_date AS TIMESTAMP) - INTERVAL '90' DAY
        AND st.created_at <= CAST(ca.ref_end_date AS TIMESTAMP) + INTERVAL '1' DAY - INTERVAL '1' SECOND
    GROUP BY ca.account_id
),
retained_ticket_counts AS (
    SELECT
        ra.account_id,
        COUNT(DISTINCT st.ticket_id) AS ticket_count
    FROM retained_accounts ra
    LEFT JOIN saasco_staging.stg_support_tickets st
        ON st.account_id = ra.account_id
        AND st.created_at >= TIMESTAMP '2026-04-01 00:00:00'
        AND st.created_at <= TIMESTAMP '2026-06-30 23:59:59'
    GROUP BY ra.account_id
)
SELECT
    'churned_q2_2026' AS group_label,
