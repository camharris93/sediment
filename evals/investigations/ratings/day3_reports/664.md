```sql
WITH max_start AS (
    SELECT MAX(start_date) AS max_start_date
    FROM saasco_marts.mart_subscription_lifecycle
),
filtered_subs AS (
    SELECT
        s.subscription_id,
        s.start_date,
        s.end_date,
        s.ended_by,
        s.first_8wk_peak_active_users,
        CASE
            WHEN s.ended_by = 'churn'
                 AND s.end_date IS NOT NULL
                 AND DATEDIFF('day', s.start_date, s.end_date) <= 180
            THEN 1
            ELSE 0
        END AS churned_180d
    FROM saasco_marts.mart_subscription_lifecycle s
    CROSS JOIN max_start m
    WHERE s.start_date <= m.max_start_date - INTERVAL '6 months'
),
tiered AS (
    SELECT
        fs.subscription_id,
        fs.churned_180d,
        NTILE(3) OVER (ORDER BY fs.first_8wk_peak_active_users) AS ntile_bucket
    FROM filtered_subs fs
),
labeled AS (
    SELECT
        t.subscription_id,
        t.churned_180d,
        CASE t.ntile_bucket
            WHEN 1 THEN 'low'
            WHEN 2 THEN 'medium'
            WHEN 3 THEN 'high'
        END AS engagement_tier,
        t.ntile_bucket
    FROM tiered t
)
SELECT
    l.engagement_tier,
    COUNT(l.subscription_id)                        AS n_subscriptions,
    ROUND(AVG(l.churned_180d::DOUBLE), 4)           AS churned_within_180d_rate
FROM labeled l
GROUP BY l.engagement_tier, l.ntile_bucket
ORDER BY l.ntile_bucket
```

```sql
WITH account_base AS (
    SELECT
        a.account_id,
        a.signup_date,
        MIN(sl.start_date) AS first_sub_start_date
    FROM saasco_staging.stg_accounts AS a
    JOIN saasco_marts.mart_subscription_lifecycle AS sl
        ON a.account_id = sl.account_id
    WHERE a.signup_date < DATE '2026-05-01'
    GROUP BY a.account_id, a.signup_date
),
churn_flag AS (
    SELECT
        sl.account_id,
        MAX(CASE
            WHEN sl.ended_by = 'churn'
             AND sl.end_date IS NOT NULL
             AND (sl.end_date - sl.start_date) <= 183
            THEN 1 ELSE 0
        END) AS churned_within_6m_flag
    FROM saasco_marts.mart_subscription_lifecycle AS sl
    GROUP BY sl.account_id
),
account_with_churn AS (
    SELECT
        ab.account_id,
        ab.first_sub_start_date,
        CASE WHEN cf.churned_within_6m_flag = 1 THEN 'yes' ELSE 'no' END AS churned_within_6m
    FROM account_base AS ab
    JOIN churn_flag AS cf
        ON ab.account_id = cf.account_id
),
usage_first_8wk AS (
    SELECT
        u.account_id,
        AVG(u.key_actions) AS avg_weekly_key_actions
    FROM saasco_staging.stg_usage_weekly AS u
    JOIN account_with_churn AS awc
        ON u.account_id = awc.account_id
    WHERE u.week_start >= awc.first_sub_start_date
      AND u.week_start <= awc.first_sub_start_date + INTERVAL '55' DAY
    GROUP BY u.account_id
)
SELECT
    awc.churned_within_6m,
    COUNT(DISTINCT awc.account_id) AS n_accounts,
    AVG(u8.avg_weekly_key_actions) AS avg_weekly_key_actions_first_8wk
FROM account_with_churn AS awc
JOIN usage_first_8wk AS u8
    ON awc.account_id = u8.account_id
GROUP BY awc.churned_within_6m
ORDER BY awc.churned_within_6m
```

```sql
WITH cohort_accounts AS (
    SELECT
        account_id,
        channel,
        MAX(CASE WHEN is_churn_month = TRUE AND tenure_month <= 6 THEN 1 ELSE 0 END) AS is_early_churner
    FROM saasco_marts.mart_account_months
    WHERE cohort_month < '2026-05-01'
    GROUP BY account_id, channel
)
SELECT
    channel,
    COUNT(DISTINCT account_id) AS n_accounts,
    ROUND(
        SUM(is_early_churner)::DOUBLE / NULLIF(COUNT(DISTINCT account_id), 0),
        4
    ) AS early_churn_rate
FROM cohort_accounts
GROUP BY channel
ORDER BY channel
```

```sql
WITH base AS (
    SELECT
        sl.subscription_id,
        sl.account_id,
        sl.plan_tier,
        sl.months_active,
        sl.ended_by,
        a.acquisition_channel AS channel,
        DATE_TRUNC('month', a.signup_date) AS cohort_month
    FROM saasco_marts.mart_subscription_lifecycle AS sl
    JOIN saasco_staging.stg_accounts AS a
        ON sl.account_id = a.account_id
    WHERE DATE_TRUNC('month', a.signup_date) < DATE '2026-05-01'
      AND sl.plan_tier IN ('starter', 'growth', 'pro')
),
aggregated AS (
    SELECT
        base.channel,
        base.plan_tier,
        COUNT(DISTINCT base.account_id) AS n_accounts,
        ROUND(
            SUM(CASE WHEN base.ended_by = 'churn' AND base.months_active <= 6 THEN 1 ELSE 0 END)::DOUBLE
            / NULLIF(COUNT(DISTINCT base.account_id), 0),
            4
        ) AS early_churn_rate
    FROM base
    GROUP BY base.channel, base.plan_tier
)
SELECT
    aggregated.channel,
    aggregated.plan_tier,
    aggregated.n_accounts,
    aggregated.early_churn_rate
FROM aggregated
ORDER BY aggregated.channel, aggregated.plan_tier
```

<details><summary>SQL behind this finding</summary>

```sql
SELECT
    CASE
        WHEN failed_attempts_before_end = 0 THEN '0'
        WHEN failed_attempts_before_end = 1 THEN '1'
        WHEN failed_attempts_before_end = 2 THEN '2'
        WHEN failed_attempts_before_end = 3 THEN '3'
        ELSE '4+'
    END AS failed_attempts_bucket,
    COUNT(*) AS n_subscriptions,
    ROUND(AVG(churned_within_90d::INT), 4) AS early_churn_rate
FROM saasco_marts.mart_subscription_lifecycle
WHERE end_date IS NOT NULL
  AND failed_attempts_before_end IS NOT NULL
GROUP BY failed_attempts_bucket
ORDER BY
    CASE failed_attempts_bucket
        WHEN '0'  THEN 0
        WHEN '1'  THEN 1
        WHEN '2'  THEN 2
        WHEN '3'  THEN 3
        ELSE 4
    END
```

```sql
WITH first_subs AS (
    SELECT
        account_id,
        subscription_id,
        start_date,
        end_date,
        ended_by,
        CASE
            WHEN end_date IS NOT NULL
                AND ended_by = 'churn'
                AND end_date <= start_date + INTERVAL '6 months'
            THEN 'yes'
            ELSE 'no'
        END AS churned_within_6m
    FROM saasco_marts.mart_subscription_lifecycle
    WHERE start_date < DATE '2026-05-01'
    QUALIFY ROW_NUMBER() OVER (PARTITION BY account_id ORDER BY start_date ASC, subscription_id ASC) = 1
),
failed_in_first_6m AS (
    SELECT
        i.account_id,
        fs.subscription_id
    FROM saasco_staging.stg_invoices i
    JOIN first_subs fs
        ON i.subscription_id = fs.subscription_id
    WHERE i.status = 'failed'
      AND i.invoice_date < fs.start_date + INTERVAL '6 months'
    GROUP BY i.account_id, fs.subscription_id
),
account_level AS (
    SELECT
        fs.account_id,
        fs.churned_within_6m AS churned_within_6m,
        CASE WHEN ff.account_id IS NOT NULL THEN 1 ELSE 0 END AS has_failed_payment
    FROM first_subs fs
    LEFT JOIN failed_in_first_6m ff
        ON fs.account_id = ff.account_id
        AND fs.subscription_id = ff.subscription_id
)
SELECT
    al.churned_within_6m,
    COUNT(*) AS n_accounts,
    ROUND(100.0 * SUM(al.has_failed_payment) / COUNT(*), 2) AS pct_with_failed_payment
FROM account_level al
GROUP BY al.churned_within_6m
ORDER BY al.churned_within_6m DESC
```

```sql
SELECT
    CAST(EXTRACT(YEAR FROM start_date) AS VARCHAR)
        || '-Q'
        || CAST(EXTRACT(QUARTER FROM start_date) AS VARCHAR) AS cohort_quarter,
    COUNT(*) AS n_subscriptions,
    ROUND(
        SUM(CASE WHEN churned_within_90d THEN 1.0 ELSE 0.0 END) / COUNT(*),
        4
    ) AS early_churn_rate_90d
FROM saasco_marts.mart_subscription_lifecycle
WHERE start_date BETWEEN DATE '2024-01-01' AND DATE '2026-06-30'
GROUP BY
    EXTRACT(YEAR FROM start_date),
    EXTRACT(QUARTER FROM start_date),
    cohort_quarter
ORDER BY
    EXTRACT(YEAR FROM start_date),
    EXTRACT(QUARTER FROM start_date)
```
