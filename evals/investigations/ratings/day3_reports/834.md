```sql
SELECT
    plan_tier,
    period,
    SUM(monthly_churned)    AS churned_accounts,
    SUM(monthly_starting)   AS starting_accounts,
```

```sql
WITH churned AS (
    SELECT
        am.plan_tier,
        CASE
            WHEN am.month IN (DATE '2026-01-01', DATE '2026-02-01', DATE '2026-03-01') THEN 'Q1 2026'
            WHEN am.month IN (DATE '2026-04-01', DATE '2026-05-01', DATE '2026-06-01') THEN 'Q2 2026'
        END AS period,
        COUNT(DISTINCT am.account_id) AS churned_accounts
    FROM saasco_marts.mart_account_months AS am
    WHERE
        am.is_churn_month = TRUE
        AND am.month IN (
            DATE '2026-01-01', DATE '2026-02-01', DATE '2026-03-01',
            DATE '2026-04-01', DATE '2026-05-01', DATE '2026-06-01'
        )
    GROUP BY am.plan_tier, period
),
starting AS (
    SELECT
        am.plan_tier,
        'Q1 2026' AS period,
        COUNT(DISTINCT am.account_id) AS starting_accounts
    FROM saasco_marts.mart_account_months AS am
    WHERE am.month = DATE '2026-01-01'
    GROUP BY am.plan_tier

    UNION ALL

    SELECT
        am.plan_tier,
        'Q2 2026' AS period,
        COUNT(DISTINCT am.account_id) AS starting_accounts
    FROM saasco_marts.mart_account_months AS am
    WHERE am.month = DATE '2026-04-01'
    GROUP BY am.plan_tier
)
SELECT
    s.plan_tier,
    s.period,
    s.starting_accounts,
    COALESCE(c.churned_accounts, 0) AS churned_accounts,
    ROUND(
        COALESCE(c.churned_accounts, 0) * 1.0 / NULLIF(s.starting_accounts, 0),
        4
    ) AS logo_churn_rate
FROM starting s
LEFT JOIN churned c
    ON s.plan_tier = c.plan_tier
    AND s.period = c.period
ORDER BY s.period, s.plan_tier
```

```sql
WITH churn_events AS (
    SELECT
        account_id,
        MAX(month) AS churn_month
    FROM saasco_marts.mart_account_months
    WHERE is_churn_month = TRUE
    GROUP BY account_id
),

churn_with_period AS (
    SELECT
        ce.account_id,
        ce.churn_month,
        CASE
            WHEN YEAR(ce.churn_month) = 2026 AND MONTH(ce.churn_month) BETWEEN 4 AND 6
                THEN 'Q2_2026'
            ELSE 'earlier'
        END AS churn_period
    FROM churn_events ce
),

pre_churn_months AS (
    SELECT
        cwp.account_id,
        cwp.churn_month,
        cwp.churn_period,
        offsets.months_before_churn,
        DATE_TRUNC('month', cwp.churn_month - INTERVAL (offsets.months_before_churn) MONTH) AS pre_churn_month
    FROM churn_with_period cwp
    CROSS JOIN (
        SELECT 1 AS months_before_churn
        UNION ALL SELECT 2
        UNION ALL SELECT 3
    ) AS offsets
),

usage_matched AS (
    SELECT
        pcm.account_id,
        pcm.churn_period,
        pcm.months_before_churn,
        u.week_start,
        u.key_actions
    FROM pre_churn_months pcm
    JOIN saasco_staging.stg_usage_weekly u
        ON u.account_id = pcm.account_id
        AND DATE_TRUNC('month', u.week_start) = pcm.pre_churn_month
),

account_week_agg AS (
    SELECT
        um.churn_period,
        um.months_before_churn,
        um.account_id,
        um.week_start,
        AVG(um.key_actions) AS avg_key_actions_week
    FROM usage_matched um
    GROUP BY um.churn_period, um.months_before_churn, um.account_id, um.week_start
),

final_agg AS (
    SELECT
        awa.churn_period,
        awa.months_before_churn,
        COUNT(DISTINCT awa.account_id) AS n_accounts,
        AVG(awa.avg_key_actions_week) AS avg_weekly_key_actions
    FROM account_week_agg awa
    GROUP BY awa.churn_period, awa.months_before_churn
)

SELECT
    fa.churn_period,
    fa.months_before_churn,
    fa.n_accounts,
    ROUND(fa.avg_weekly_key_actions, 4) AS avg_weekly_key_actions
FROM final_agg fa
ORDER BY fa.churn_period, fa.months_before_churn
```
