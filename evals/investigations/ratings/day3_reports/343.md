WITH monthly_agg AS (
    SELECT
        am.month,
        am.cohort_month,
        am.plan_tier,
        am.channel,
        CASE
            WHEN EXTRACT(YEAR FROM am.month) = 2024 THEN '2024'
            ELSE '2025_2026'
        END AS year_era,
        COUNT(*) AS starting_accounts,
        SUM(CASE WHEN am.is_churn_month = TRUE THEN 1 ELSE 0 END) AS churned_accounts,
        SUM(CASE WHEN am.is_churn_month = TRUE THEN am.mrr ELSE 0 END) AS churned_mrr
    FROM saasco_marts.mart_account_months AS am
    WHERE
        am.region = 'EU'
        AND am.month >= DATE '2024-01-01'
        AND am.month <= DATE '2026-06-01'
    GROUP BY
        am.month,
        am.cohort_month,
        am.plan_tier,
        am.channel,
        CASE
            WHEN EXTRACT(YEAR FROM am.month) = 2024 THEN '2024'
            ELSE '2025_2026'
        END
)
SELECT
    ma.month,
    ma.cohort_month,
    ma.plan_tier,
    ma.channel,
    ma.year_era,
    ma.starting_accounts,
    ma.churned_accounts,
    ma.churned_mrr,
    CASE
        WHEN ma.starting_accounts > 0
        THEN ROUND(ma.churned_accounts::DOUBLE / ma.starting_accounts, 4)
        ELSE NULL
    END AS logo_churn_rate,
    mk.logo_churn_rate AS company_total_logo_churn_rate,
    mk.churned_mrr AS company_total_churned_mrr
FROM monthly_agg AS ma
LEFT JOIN saasco_marts.mart_monthly_kpis AS mk
    ON mk.month = ma.month
ORDER BY
    ma.month ASC,
    ma.plan_tier ASC,
    ma.channel ASC,
    ma.cohort_month ASC

```sql
WITH eu_churned AS (
    SELECT
        msl.subscription_id,
        msl.account_id,
        msl.end_date,
        YEAR(msl.end_date)                          AS churn_year,
        msl.plan_tier,
        msl.billing_interval,
        msl.failed_attempts_before_end,
        msl.churned_within_90d,
        COALESCE(ss.cancel_reason, 'unknown')       AS cancel_reason
    FROM saasco_marts.mart_subscription_lifecycle   AS msl
    JOIN saasco_staging.stg_accounts                AS sa
        ON msl.account_id = sa.account_id
    JOIN saasco_staging.stg_subscriptions           AS ss
        ON msl.subscription_id = ss.subscription_id
    WHERE sa.region = 'EU'
      AND msl.ended_by = 'churn'
      AND msl.end_date IS NOT NULL
),

year_totals AS (
    SELECT
        churn_year,
        COUNT(*)                                    AS year_total_churned,
        COUNT(*) FILTER (WHERE churned_within_90d)  AS year_total_early_churn
    FROM eu_churned
    GROUP BY churn_year
),

segment_agg AS (
    SELECT
        ec.churn_year,
        ec.cancel_reason,
        ec.plan_tier,
        ec.billing_interval,
        COUNT(*)                                                        AS churned_subscriptions,
        AVG(ec.failed_attempts_before_end)                              AS avg_failed_attempts,
        COUNT(*) FILTER (WHERE ec.churned_within_90d)                   AS early_churn_count,
        ROUND(
            100.0 * COUNT(*) FILTER (WHERE ec.churned_within_90d)
            / NULLIF(COUNT(*), 0),
            2
        )                                                               AS early_churn_rate_pct
    FROM eu_churned AS ec
    GROUP BY
        ec.churn_year,
        ec.cancel_reason,
        ec.plan_tier,
        ec.billing_interval
)

SELECT
    sa.churn_year,
    sa.cancel_reason,
    sa.plan_tier,
    sa.billing_interval,
    sa.churned_subscriptions,
    ROUND(
        100.0 * sa.churned_subscriptions / NULLIF(yt.year_total_churned, 0),
        2
    )                                                                   AS pct_of_year_by_segment,
    ROUND(
        100.0 * SUM(sa.churned_subscriptions) OVER (
            PARTITION BY sa.churn_year, sa.cancel_reason
        ) / NULLIF(yt.year_total_churned, 0),
        2
    )                                                                   AS cancel_reason_pct_of_year,
    ROUND(
        100.0 * SUM(sa.churned_subscriptions) OVER (
            PARTITION BY sa.churn_year, sa.plan_tier
        ) / NULLIF(yt.year_total_churned, 0),
        2
    )                                                                   AS plan_tier_pct_of_year,
    ROUND(
        100.0 * SUM(sa.churned_subscriptions) OVER (
            PARTITION BY sa.churn_year, sa.billing_interval
        ) / NULLIF(yt.year_total_churned, 0),
        2
    )                                                                   AS billing_interval_pct_of_year,
    yt.year_total_churned,
    yt.year_total_early_churn,
    ROUND(
        100.0 * yt.year_total_early_churn / NULLIF(yt.year_total_churned, 0),
        2
    )                                                                   AS year_early_churn_rate_pct,
    sa.early_churn_count,
    sa.early_churn_rate_pct,
    ROUND(sa.avg_failed_attempts, 4)                                    AS avg_failed_attempts_before_end
FROM segment_agg AS sa
JOIN year_totals AS yt
    ON sa.churn_year = yt.churn_year
ORDER BY
    sa.churn_year,
    sa.cancel_reason,
    sa.plan_tier,
    sa.billing_interval
```
